@login_required
def get_report_forms(request):
    form = ReportForm(request.POST or None)

    if request.method == "POST" and form.is_valid():
        grade = form.cleaned_data["grade"]

        exam_codes = (
            form.cleaned_data["opener"].exam_code,
            form.cleaned_data["midterm"].exam_code,
            form.cleaned_data["endterm"].exam_code,
        )

        filename = "PLAY.docx"
        template_path = BASE_DIR / filename

        placeholders = get_placeholders(docx_path=template_path)

        reports = lookup(exam_codes, grade, placeholders)

        file = generate_class_reports(template_path, reports)

        return FileResponse(
            file,
            as_attachment=True,
            filename=f"{grade.name} Report Forms.docx"
        )


    return; return render(request, "get_reportforms.html", {
        "form": form
    })


def lookup(exam_codes, grade, placeholders):
    learners = list(Learner.objects.filter(grade=grade))
    exams = list(Exam.objects.filter(grade=grade, exam_code__in=exam_codes).select_related("subject"))
    marks = list(Mark.objects.filter(student__in=learners, exam__in=exams))

    exam_dict = {(exam.exam_code, exam.subject.code): exam.id for exam in exams}
    mark_dict = {(mark.student_id, mark.exam_id): mark.score for mark in marks}

    reports = {}

    for learner in learners:
        data = {}

        for ph in placeholders:
            if ph == "NAME":
                data[ph] = learner.name
                continue

            if ph == "GRADE":
                data[ph] = learner.grade.name
                continue

            if ph == "YEAR":
                data[ph] = 2026
                continue

            if "_" not in ph:
                continue

            subject_code, term = ph.split("_")

            if term == "0":
                scores = []

                for exam_code in exam_codes:
                    exam_id = exam_dict.get((exam_code, subject_code))

                    if exam_id:
                        scores.append(
                            mark_dict.get((learner.id, exam_id), 0)
                        )

                data[ph] = round(sum(scores) / len(scores), 2) if scores else ""
                continue

            if term == "1":
                exam_code = exam_codes[0]

            elif term == "2":
                exam_code = exam_codes[1]

            elif term == "3":
                exam_code = exam_codes[2]

            else:
                data[ph] = ""
                continue

            exam_id = exam_dict.get((exam_code, subject_code))

            data[ph] = mark_dict.get((learner.id, exam_id), "") if exam_id else ""

        reports[learner] = data

    print(reports)
    return reports


def replace_placeholders(doc, values):
    for paragraph in doc.paragraphs:
        for key, value in values.items():
            placeholder = f"[{key}]"

            if placeholder in paragraph.text:
                for run in paragraph.runs:
                    run.text = run.text.replace(placeholder, str(value))

    for table in doc.tables:
        for row in table.rows:
            for cell in row.cells:
                for paragraph in cell.paragraphs:
                    for key, value in values.items():
                        placeholder = f"[{key}]"

                        if placeholder in paragraph.text:
                            for run in paragraph.runs:
                                run.text = run.text.replace(placeholder, str(value))


def generate_class_reports(template_path, reports):
    master = Document()

    if master.paragraphs:
        p = master.paragraphs[0]._element
        p.getparent().remove(p)

    for index, values in enumerate(reports.values()):
        doc = Document(template_path)

        replace_placeholders(doc, values)

        for element in doc.element.body:
            master.element.body.append(deepcopy(element))

        if index < len(reports) - 1:
            master.add_page_break()

    buffer = BytesIO()
    master.save(buffer)
    buffer.seek(0)

    return buffer

import re
from docx import Document

PLACEHOLDER_PATTERN = re.compile(r"\[([A-Za-z0-9_]+)\]")

def get_placeholders(docx_path=""):
    """
    Extract all placeholders like [NAME], [MAT_1], [TEST_0]
    from a .docx file.

    Returns:
        list[str]: Unique placeholders (without brackets),
                   in the order they first appear.
    """
    if not docx_path:
        return []
    doc = Document(docx_path)
    placeholders = []

    def extract(text):
        for match in PLACEHOLDER_PATTERN.findall(text):
            if match not in placeholders:
                placeholders.append(match)

    # Normal paragraphs
    for paragraph in doc.paragraphs:
        extract(paragraph.text)

    # Tables
    for table in doc.tables:
        for row in table.rows:
            for cell in row.cells:
                extract(cell.text)

    return placeholders

from django.shortcuts import render, redirect, get_object_or_404

from django.core.paginator import Paginator
from django.contrib import messages
from .models import Learner, Grade
from accounts.models import User
import uuid


@login_required
def add_student(request):
    s_id = uuid.uuid4().hex[:12].upper()
    if request.user.role != 'admin':
        return redirect('dashboard')

    grades = Grade.objects.all()  # Fetch all grades from DB

    if request.method == "POST":
        name = request.POST.get("name")
        admission_number = request.POST.get("admission_number")
        grade_id = request.POST.get("grade")

        if not (name and admission_number and grade_id):
            messages.error(request, "All fields are required.")
        else:
            try:
                grade_obj = get_object_or_404(Grade, id=grade_id)
                learner = Learner.objects.create(
                    name=name,
                    admission_number=admission_number,
                    grade=grade_obj
                )
                learner.save()
                messages.success(request, f"Learner {name} added successfully!")
                return redirect('add_learner')
            except Grade.DoesNotExist:
                messages.error(request, "Selected grade does not exist.")
            except Exception as e:
                messages.error(request, f"Error adding learner: {str(e)}")

    return render(request, 'add_student.html', {
        "lid": s_id,
        "grades": grades
    })


@login_required
def grade_list(request):
    if request.user.role != 'admin':
        return redirect('dashboard')

    grades = Grade.objects.all()

    return render(request, 'list_grades.html', {

@login_required
def add_grade(request):
    if request.user.role != 'admin':
        return redirect('dashboard')

    teachers = User.objects.filter(role='teacher')

    if request.method == "POST":
        name = request.POST.get("name")
        teacher_id = request.POST.get("class_teacher")

        if not name:
            messages.error(request, "Grade name is required.")
        else:
            teacher = None
            if teacher_id:
                teacher = User.objects.filter(id=teacher_id).first()

            Grade.objects.create(
                name=name,
                class_teacher=teacher
            )

            messages.success(request, f"Grade {name} created successfully!")
            return redirect('grade_list')

    return render(request, 'add_grade.html', {
        "teachers": teachers
    })

@login_required
def grade_detail(request, grade_id):
    if request.user.role != 'admin':
        return redirect('dashboard')

    grade = get_object_or_404(Grade, id=grade_id)
    learners = Learner.objects.select_related('grade').all()
    learners = learners.filter(grade=grade)
    teachers = User.objects.filter(role='teacher')

    if request.method == "POST":
        teacher_id = request.POST.get("class_teacher")

        if teacher_id:
            grade.class_teacher = get_object_or_404(User, id=teacher_id)
        else:
            grade.class_teacher = None

        grade.save()
        messages.success(request, "Class teacher updated.")
        return redirect('grade_detail', grade_id=grade.id)

    return render(request, 'grade_details.html', {
        'grade': grade,
        'learners': learners,
        'teachers': teachers
    })


@login_required(login_url="/login/")
def learners_list(request):
    grades = Grade.objects.all()
    learners = Learner.objects.select_related('grade').all()

    # 🔍 FILTERS
    grade_id = request.GET.get('grade', '')
    learner_name = request.GET.get('learner_name', '')
    print(learner_name)

    if grade_id:
        learners = learners.filter(grade_id=grade_id)

    if learner_name:
        learners = learners.filter(name__icontains=learner_name)

    # 📄 PAGINATION (10 per page)
    paginator = Paginator(learners, 10)
    page_number = request.GET.get('page', 1)
    page_obj = paginator.get_page(page_number)


    context = {
        'page_obj': page_obj,
        'grades': grades,
        'filters': {
            'selected_grade': grade_id or '',
            'learner_name': learner_name or ''
        }
    }
    return render(request, 'list_learners.html', context)

from django import forms
from .models import User
from django.conf import settings


class RegisterForm(forms.ModelForm):
    password = forms.CharField(widget=forms.PasswordInput)
    admin_key = forms.CharField(required=False)

    class Meta:
        model = User
        fields = ['username', 'password', 'role']

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)

        for field in self.fields.values():
            field.widget.attrs.update({
                'class': 'w-full mt-1 p-2 border rounded focus:ring-2 focus:ring-green-500 focus:outline-none'
            })

    def clean(self):
        cleaned_data = super().clean()
        role = cleaned_data.get('role')
        admin_key = cleaned_data.get('admin_key')

        if role == 'admin':
            if admin_key != settings.ADMIN_CREATION_KEY:
                raise forms.ValidationError("Invalid admin pass key")

        return cleaned_data

def reset_password(request):
    if request.method == "POST":
        username = request.POST.get("username")
        new_password = request.POST.get("new_password")
        admin_key = request.POST.get("admin_key")

        # check admin key first
        if admin_key != settings.ADMIN_CREATION_KEY:
            return render(request, "reset_password.html", {
                "error": "Invalid admin pass key"
            })
        try:
            user = User.objects.get(username=username)
            user.password = make_password(new_password)
            user.save()
            messages.success(request, "Password reset successful. Please log in.")
            return redirect("login")
        except User.DoesNotExist:
            return render(request, "reset_password.html", {
                "error": "User not found"
            })
    return render(request, "reset_password.html")

def close_expired_exam_codes():
    cutoff_date = timezone.now().date() - timedelta(days=14)

    ExamCode.objects.filter(
        is_open=True,
        date_taken__lt=cutoff_date
    ).update(is_open=False)


def register(request):
    if request.user.is_authenticated:
        return redirect('dashboard')

    if request.method == 'POST':
        form = RegisterForm(request.POST)
        if form.is_valid():
            user = form.save(commit=False)
            user.set_password(form.cleaned_data['password'])
            user.save()

            login(request, user)
            messages.success(request, "Account created successfully.")
            return redirect('dashboard')
    else:
        form = RegisterForm()

    return render(request, 'register.html', {'form': form})


def login_view(request):
    if request.user.is_authenticated:
        return redirect('dashboard')

    error = None

    if request.method == 'POST':
        username = request.POST.get('username')
        password = request.POST.get('password')

        user = authenticate(username=username, password=password)

        if user:
            login(request, user)
            messages.success(request, "Account logged in successfully.")
            return redirect('dashboard')
        else:
            error = "Invalid username or password"

    return render(request, 'login.html', {'error': error})

@login_required(login_url="/login/")
def teacher(request):
    exams = Exam.objects.select_related('grade').order_by('-date_taken')

    # 🔍 FILTERS
    code = request.GET.get('code')
    grade = request.GET.get('grade')
    subject = request.GET.get('subject')

    if code:
        exams = exams.filter(code_id=code)

    if grade:
        exams = exams.filter(grade_id=grade)

    if subject:
        exams = exams.filter(subject_name__icontains=subject)

    if request.user.role != "admin":
        exams = exams.filter(created_by=request.user)

    # 📄 PAGINATION (10 per page)
    paginator = Paginator(exams, 10)
    page_number = request.GET.get('page')
    page_obj = paginator.get_page(page_number)

    # 📦 GROUP BY exam_code
    grouped_exams = defaultdict(list)
    for exam in page_obj:
        grouped_exams[exam.exam_code].append(exam)

    context = {
        'page_obj': page_obj,
        'grouped_exams': dict(grouped_exams),
        'codes': ExamCode.objects.filter(is_open=True),
        'grades': Grade.objects.all(),
        'filters': {
            'code': code or '',
            'grade': grade or '',
            'subject': subject or ''
        }
    }

    return render(request, 'teacher.html', context)

from django.db.models.signals import post_save, post_delete
from django.dispatch import receiver
from django.db import transaction
from exams.models import Mark, Exam
from exams.services.recalculate import recalc_exam
from exams.signal_state import signals_enabled


def trigger_recalc(raw_exam):
    # 🚨 prevent recursion trigger
    if not signals_enabled():
        return

    final_exams = Exam.objects.filter(
        is_final=True,
        depends_on=raw_exam
    )

    for final_exam in final_exams:
        # ensure DB commit before recalculation
        transaction.on_commit(lambda e=final_exam: recalc_exam(e))


@receiver(post_save, sender=Mark)
def mark_saved(sender, instance, **kwargs):
    trigger_recalc(instance.exam)


@receiver(post_delete, sender=Mark)
def mark_deleted(sender, instance, **kwargs):
    trigger_recalc(instance.exam)

@login_required
def double_class_results(request):
    """
    Page 2: Show Class Marklist results for two classes
    """
    data = request.session.get("double_class_data")
    if not data:
        return redirect("double")

    first_grade = get_object_or_404(
        Grade,
        id=data["first_grade"]
    )

    second_grade = get_object_or_404(
        Grade,
        id=data["second_grade"]
    )

    exam_code = data["exam_code"]
    show_points = data.get("show_points_only", False)

    # Fetch exams for both classes
    exams = Exam.objects.filter(
        grade__in=[first_grade, second_grade],
        is_final=True
    ).filter(
        models.Q(exam_code=exam_code) |
        models.Q(code__exam_code=exam_code)
    )

    # Unique subjects for table columns
    subjects = list(
        exams.values_list(
            "subject_name",
            flat=True
        ).distinct()
    )
    subjects = [
        subject.code
        for subject in Subject.objects.filter(exam__in=exams).distinct().order_by("priority")
    ]

    # All learners from both classes
    learners = Learner.objects.filter(
        grade__in=[first_grade, second_grade]
    ).select_related("grade")

    # Initialize totals
    subject_totals = {subject: 0 for subject in subjects}

    # Fetch all marks once
    all_marks = Mark.objects.filter(
        student__in=learners,
        exam__in=exams
    )

    # Fast lookup for marks
    mark_dict = {
        (m.student_id, m.exam_id): int(m.score)
        for m in all_marks
    }

    # Fast lookup for exams
    exam_lookup = {
        (exam.grade_id, exam.subject.code): exam.id
        for exam in exams
    }

    results = []

    for student in learners:
        student_data = {
            'student': student,
            'scores': {},
            'total': 0
        }

        total = 0

        for subject in subjects:

            # Get the correct exam for this student's class
            exam_id = exam_lookup.get(
                (student.grade_id, subject)
            )

            score = 0
            if exam_id:
                score = mark_dict.get(
                    (student.id, exam_id),
                    0
                )

            value = get_points(score) if show_points else score

            student_data['scores'][subject] = value

            total += value

            subject_totals[subject] += (
                value if show_points else score
            )

        student_data['total'] = total
        student_data['average'] = (
            total / len(subjects)
            if subjects else 0
        )

        results.append(student_data)

    # Compute means
    num_students = len(learners)

    subject_means = {
        subject: round(
            subject_totals[subject] / num_students,
            2
        ) if num_students > 0 else 0
        for subject in subjects
    }

    # Class stats
    class_total = sum(
        r['total']
        for r in results
    )

    class_mean = round(
        class_total / len(results),
        2
    ) if results else 0

    # Sort by total descending
    results.sort(
        key=lambda x: x['total'],
        reverse=True
    )

    return render(request, 'class_marklist.html', {
        'results': results,
        'subjects': subjects,
        'selected_grade': f"{first_grade.name} and {second_grade.name}",
        'exam_code': exam_code,
        'subject_totals': subject_totals,
        'subject_means': subject_means,
        'class_total': class_total,
        'class_mean': class_mean,
        'double': True,
        'show_points': show_points
    })

@login_required
def create_exam(request):
    if request.method == 'POST':
        form = ExamForm(request.POST)
        if form.is_valid():
            exam = form.save(commit=False)
            #if int(exam.max_score) == int(100):
            exam.is_final = True

            exam.created_by = request.user
            existing_exam = Exam.objects.filter(
                exam_code=exam.code.exam_code,
                grade=exam.grade,
                #subject_name=exam.subject_name
                subject_name=exam.subject.code
            ).first()

            if existing_exam:
                messages.success(request, "Exam already exists. Opening marks entry...")
                return redirect('enter_marks', exam_id=existing_exam.id)
            else:
                exam.save()
                return redirect('enter_marks', exam_id=exam.id)
        else:
            messages.error(request, form.errors)
            code_present = ExamCode.objects.filter(is_open=True).exists()
            return render(request, 'create_exam.html', {'form': form, 'code_present': code_present})
    else:
        form = ExamForm()
        code_present = ExamCode.objects.filter(is_open=True).exists()
    return render(request, 'create_exam.html', {'form': form, 'code_present': code_present})

import ast
import operator

# Supported operators
OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Pow: operator.pow,
}

def safe_eval(expr, variables):
    """
    Safely evaluate a math expression using AST
    """
    def _eval(node):
        if isinstance(node, ast.Num):  # number
            return node.n

        elif isinstance(node, ast.Name):  # variable
            return variables.get(node.id, 0)

        elif isinstance(node, ast.BinOp):  # x + y
            left = _eval(node.left)
            right = _eval(node.right)
            return OPERATORS[type(node.op)](left, right)

        elif isinstance(node, ast.UnaryOp):  # -x
            return -_eval(node.operand)

        else:
            raise ValueError("Unsupported expression")

    tree = ast.parse(expr, mode='eval')
    return _eval(tree.body)

class Subject(models.Model):
    code = models.CharField(max_length=10)
    name = models.CharField(max_length=50)
    priority = models.PositiveIntegerField()

    class Meta:
        ordering = ["priority"]
        unique_together = ('code', 'name')

    def __str__(self):
        return f'{self.name} ({self.code})'

class Exam(models.Model):
    exam_code = models.CharField(max_length=20, null=True)  # old data
    code = models.ForeignKey(ExamCode, on_delete=models.CASCADE, blank=True, null=True) #backward compatibility
    grade = models.ForeignKey(Grade, on_delete=models.CASCADE)
    subject_name = models.CharField(max_length=32)
    subject = models.ForeignKey(Subject, on_delete=models.PROTECT, blank=True, null=True) # backward compatibility
    max_score = models.FloatField(blank=True, null=True, default=100)
    date_taken = models.DateField()
    created_by = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE)
    is_final = models.BooleanField(default=False)  # ✅ NEW
    formula = models.TextField(blank=True, null=True)
    depends_on = models.ManyToManyField("self", blank=True, symmetrical=False)

    class Meta:
        unique_together = ('exam_code', 'grade', 'subject')

    @property
    def resolved_exam_code(self):
        return self.code.exam_code if self.code else self.exam_code

    def save(self, *args, **kwargs):
        if self.code:
            self.exam_code = self.code.exam_code
        super().save(*args, **kwargs)

    def __str__(self):
        return f'{self.exam_code} {self.grade} {self.subject_name} out of {int(self.max_score)}'


class Mark(models.Model):
    student = models.ForeignKey(Learner, on_delete=models.CASCADE)
    exam = models.ForeignKey(Exam, on_delete=models.CASCADE)
    score = models.FloatField()

    class Meta:
        unique_together = ('student', 'exam')  # Prevent double entries

    def __str__(self):
        return f"{self.student.name} - {self.exam.exam_code} : {self.score}"

from django.db import transaction
from exams.models import Mark
from exams.utils import safe_eval
from exams.signal_state import disable_signals, enable_signals


def recalc_exam(final_exam):
    raw_exams = list(final_exam.depends_on.all())
    students = final_exam.grade.learner_set.all()

    # no need to continue if no formula
    if not final_exam.formula:
        return

    raw_marks = Mark.objects.filter(exam__in=raw_exams)

    mark_dict = {
        (m.student_id, m.exam_id): m.score
        for m in raw_marks
    }

    new_marks = []

    for student in students:
        variables = {
            ex.subject_name: mark_dict.get((student.id, ex.id), 0)
            for ex in raw_exams
        }

        try:
            score = safe_eval(final_exam.formula, variables)
        except Exception:
            score = 0

        new_marks.append(
            Mark(
                student=student,
                exam=final_exam,
                score=score
            )
        )

    with transaction.atomic():
        # 🚨 prevent signal recursion during delete + recreate
        disable_signals()

        try:
            Mark.objects.filter(exam=final_exam).delete()
            Mark.objects.bulk_create(new_marks)
        finally:
            enable_signals()
