function printMarklist() {
    window.onafterprint = () => location.reload();

    // Hide unnecessary elements
    document.querySelector("aside")?.style.setProperty("display", "none");
    document.querySelector("header")?.style.setProperty("display", "none");
    document.querySelectorAll("button, a").forEach(el => {
        el.style.display = "none";
    });

    // Keep page almost full size
    document.body.style.margin = "0";
    document.body.style.padding = "0";

    const main = document.querySelector("main");
    if (main) {
        main.style.padding = "0.25rem";
        main.style.margin = "0";
        main.style.fontSize = "0.8rem";      // 11.2px instead of 9px
        main.style.width = "100%";
    }

    // Reduce wrapper spacing only a little
    document.querySelectorAll(".overflow-x-auto").forEach(div => {
        div.style.margin = "0";
        div.style.padding = "0";
    });

    // Keep headings readable
    document.querySelectorAll("h2").forEach(h => {
        h.style.fontSize = "1rem";
        h.style.margin = "0.2rem 0";
    });

    // Table uses full width
    document.querySelectorAll("table").forEach(table => {
        table.style.width = "100%";
        table.style.borderCollapse = "collapse";
        table.style.tableLayout = "auto"; // <-- don't use fixed
        table.style.fontSize = "11px";
    });

    // Compact but not tiny
    document.querySelectorAll("th, td").forEach(cell => {
        cell.style.padding = "1.5px 3px";
        cell.style.fontSize = "11px";
        cell.style.lineHeight = "1.2";
        cell.style.whiteSpace = "nowrap";
    });

    // Slightly shorter rows
    document.querySelectorAll("tr").forEach(row => {
        row.style.height = "18px";
    });

    window.print();
}
